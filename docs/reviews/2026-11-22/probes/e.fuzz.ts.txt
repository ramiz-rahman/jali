import { it } from "vitest";
import { createJali, row, column, tabs, memoryStorage, memoryLocation, layout as L } from "packages/core/src/index.ts";
import { storeOf } from "packages/core/src/adapter.ts";

const kinds = [
  { kind: "list", title: "List", width: { min: 260, comfortable: 360 }, selects: "item" },
  { kind: "item", title: (p: any) => "Item " + p.id, params: { id: Number }, width: { min: 320, max: 700 } },
  { kind: "wide", title: "Wide", width: { min: 520 }, height: { min: 300 } },
  { kind: "note", title: "Note", scratch: true, width: { min: 200 } },
];
const launcher = { kind: "launcher", title: "New", width: { min: 240 } };
function rng(seed: number) { return () => { seed |= 0; seed = seed + 0x6D2B79F5 | 0; let t = Math.imul(seed ^ seed >>> 15, 1 | seed); t = t + Math.imul(t ^ t >>> 7, 61 | t) ^ t; return ((t ^ t >>> 14) >>> 0) / 4294967296; }; }

it("random action sequences keep invariants", () => {
  const problems: string[] = [];
  for (let seed = 1; seed <= 400; seed++) {
    const r = rng(seed); const pick = <T,>(a: T[]) => a[Math.floor(r() * a.length)]!;
    const location = memoryLocation("");
    const unstable: string[] = [];
    const jali = createJali({ panes: kinds, launcher, layout: row({kind:"list",params:{}}, {kind:"item",params:{id:1}}), url: { location, sizes: "link" },
      storage: memoryStorage(), size: { width: 1400, height: 800 }, moveFocus: false, onUnstable: (op) => unstable.push(op) });
    const store = storeOf(jali); jali.start();
    const log: string[] = [];
    for (let i = 0; i < 60; i++) {
      const panes = L.tree.panes(jali.getLayout().root);
      const p = pick(panes)?.id; const q = pick(panes)?.id;
      const ref = pick([{kind:"item",params:{id: 1 + Math.floor(r()*6)}}, {kind:"wide",params:{}}, {kind:"note",params:{n: Math.floor(r()*3)}}, {kind:"list",params:{}}]);
      const acts: [string, () => unknown][] = [
        ["openBeside", () => jali.actions.openBeside(p, ref, { edge: pick(["left","right","top","bottom", undefined] as any) })],
        ["openHere", () => jali.actions.openHere(p, ref)],
        ["addTab", () => jali.actions.addTab(p, ref)],
        ["close", () => jali.actions.close(p)],
        ["closeOthers", () => jali.actions.closeOthers(p)],
        ["undo", () => jali.actions.undo()],
        ["tuck", () => jali.actions.tuck(p)],
        ["putBack", () => jali.actions.putBack()],
        ["move", () => jali.actions.move(p, pick(["left","right","top","bottom"] as any))],
        ["swapWithNext", () => jali.actions.swapWithNext(p)],
        ["mergeInto", () => jali.actions.mergeInto(p, q)],
        ["unmerge", () => jali.actions.unmerge(p)],
        ["moveTab", () => jali.actions.moveTab(p, Math.floor(r()*4))],
        ["setPinned", () => jali.actions.setPinned(p, r() < 0.5)],
        ["toggleZoom", () => jali.actions.toggleZoom(p)],
        ["focus", () => jali.actions.focus(p)],
        ["select", () => jali.actions.select(p, {kind:"item",params:{id: 1 + Math.floor(r()*6)}})],
        ["split", () => jali.actions.split(p, pick(["right","bottom"] as any))],
        ["newTab", () => jali.actions.newTab(p)],
        ["autoLayout", () => jali.actions.autoLayout()],
        ["resize", () => store.set({ area: { width: 380 + Math.floor(r()*1800), height: 300 + Math.floor(r()*900) } })],
        ["remove", () => jali.actions.remove(p)],
        ["setKept", () => jali.actions.setKept(p, r() < 0.5)],
        ["back", () => { if (location.entries.length > 1) { location.entries.pop(); location.visit(location.entries.pop()!); } }],
        ["reload-roundtrip", () => {
          const j2 = createJali({ panes: kinds, launcher, layout: row({kind:"list",params:{}}), url: { location: memoryLocation(location.search()), sizes: "link" }, storage: memoryStorage(), size: store.get().area, moveFocus: false });
          const a = JSON.stringify(L.tree.panes(jali.getLayout().root).map(x => [x.id, x.ref])), b = JSON.stringify(L.tree.panes(j2.getLayout().root).map(x => [x.id, x.ref]));
          if (a !== b) throw new Error("link round trip differs\n" + a + "\n" + b);
          const ga = JSON.stringify(jali.model().resolved.panes.filter(x=>!x.hidden).map(x => [x.id, Math.round(x.rect.x), Math.round(x.rect.width)]));
          const gb = JSON.stringify(j2.model().resolved.panes.filter(x=>!x.hidden).map(x => [x.id, Math.round(x.rect.x), Math.round(x.rect.width)]));
          if (ga !== gb) throw new Error("link geometry differs\n" + ga + "\n" + gb);
        }],
      ];
      const [name, run] = pick(acts);
      log.push(name + "(" + p + "," + (name.startsWith("open")||name==="addTab" ? JSON.stringify(ref) : q) + ")");
      try { run(); } catch (e: any) { problems.push("seed " + seed + " step " + i + " " + name + " threw: " + e.message.split("\n").slice(0,3).join(" | ") + "\n   " + log.slice(-6).join(" ")); break; }
      // invariants
      const ids = L.tree.panes(jali.getLayout().root).map(x => x.id);
      if (new Set(ids).size !== ids.length) { problems.push("seed " + seed + " dup ids after " + name); break; }
      let m; try { m = jali.model(); } catch (e: any) { problems.push("seed " + seed + " model threw after " + name + ": " + e.message + "\n   " + log.slice(-6).join(" ")); break; }
      const bad = m.resolved.panes.find(b => !b.hidden && ![b.rect.x,b.rect.y,b.rect.width,b.rect.height].every(Number.isFinite));
      if (bad) { problems.push("seed " + seed + " NaN rect after " + name); break; }
      const f = jali.get().view.focused; if (f && !ids.includes(f)) { problems.push("seed " + seed + " focus on missing pane after " + name); break; }
      const z = jali.get().view.zoomed; if (z && !ids.includes(z)) { problems.push("seed " + seed + " zoom on missing pane after " + name); break; }
      if (!m.single) { const over = m.resolved.panes.find(b => !b.hidden && !b.clip && (b.rect.x + b.rect.width > store.get().area.width + 1 || b.rect.x < -1)); if (over) { problems.push("seed " + seed + " pane outside area after " + name + " " + JSON.stringify(over.rect) + " area " + JSON.stringify(store.get().area) + "\n   " + log.slice(-4).join(" ")); break; } }
      if (unstable.length) { problems.push("seed " + seed + " unstable " + unstable.join(",") + " after " + log.slice(-3).join(" ")); unstable.length = 0; }
    }
    jali.destroy();
  }
  const uniq = [...new Map(problems.map(p => [p.split(" after ")[0].replace(/seed \d+ (step \d+ )?/, "") + (p.match(/threw: [^|]*/)?.[0] ?? ""), p])).values()];
  console.log("problems", problems.length, "\n" + uniq.slice(0, 14).join("\n"));
});
