<Section id="workspace" label="What you get" title="What Jali does for you">
  <Path>
    <Proof id="fold" title="Panes fold when room runs out" href="/learn/fitting" more="How a layout fits" demo={<Frame layout="crowded" height={340} />}>
      Each kind of pane declares the narrowest width it works at. Drag the frame's edge: below that width a pane folds into a tab, and it comes back when there is room.
    </Proof>
    <Proof id="follow" title="The detail follows the list" href="/learn/following" more="Following a list" demo={<Spotlight layout="chain" target="follow" height={340} />}>
      Open a record from a list, and its pane shows whichever row is selected. Keep it to stay on one record.
    </Proof>
    <Proof id="link" title="The link reopens the layout" href="/guides/url-and-routers" more="Keep the layout in the URL" demo={<LinkDemo />}>
      With `urlSync()`, every change is written into the address. Open, close or resize a pane and watch it change. Share the link, and the same panes open in the same places.
    </Proof>
    <Proof id="keys" title="Every command has a key" href="/guides/commands-and-keys" more="Commands and keys" demo={<KeysDemo />}>
      Move focus, maximise, tuck, close and undo from the keyboard, and rebind or turn off any key. Click inside the panes to try them, or press a key below.
    </Proof>
  </Path>
</Section>
