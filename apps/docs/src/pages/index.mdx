<Band label="Two packages" title="The model, and React on top of it">
  <Packages>
    <Package name="@jali/core" href="/reference/core" note="No framework" exports="createUiStore, createPaneKinds, createUrlSync, boot, startUi, layout">
      The layout tree and every change to it, fitting, and a store with focus, undo, follow and unsaved work. Keyboard commands you can rebind, and ports for the address and storage.
    </Package>
    <Package name="@jali/react" href="/reference/react" note="Components and hooks" exports="Jali, JaliView, definePaneKind, usePaneState, useActiveItemPort">
      Renders the workspace: panes, dividers, tabs, the shelf and the command palette. Ships a default theme built from `--jali-*` tokens.
    </Package>
  </Packages>
</Band>
