<Section id="does" label="What it does" title="Four things, live">
  <Features>
    <Feature title="Panes fold when room runs out" href="/learn/fitting" more="How a layout fits" demo={<Frame layout="crowded" height={300} />}>
      Each kind of pane says the narrowest it works at. Drag the frame narrower and panes fold into tabs; widen it and they come back.
    </Feature>
    <Feature title="A detail follows its list" href="/learn/following" more="Following a list" demo={<Spotlight layout="chain" target="follow" height={300} />}>
      Open an order from the list, and its pane shows whichever row you select, once the selection rests.
    </Feature>
    <Feature title="The layout lives in the link" href="/guides/url-and-routers" more="Keep the layout in the URL" demo={<LinkDemo />}>
      Every change is written into the address, so a link opens the same panes in the same places, and Back steps through what you did.
    </Feature>
    <Feature title="Keyboard first" href="/guides/commands-and-keys" more="Commands and keys" demo={<KeysDemo />}>
      Every command has a key, and you can change or turn off any of them. Click inside the workspace, then try these.
    </Feature>
  </Features>
</Section>

<Section id="packages" label="Two packages" title="The model, and React on top of it">
  <Packages>
    <Package name="@jali/core" href="/reference/core" exports="createUiStore, createPaneKinds, createUrlSync, boot, startUi, layout">
      The layout tree and every change to it, fitting, and a store with focus, undo, follow and unsaved work. No framework.
    </Package>
    <Package name="@jali/react" href="/reference/react" exports="Jali, JaliView, definePaneKind, usePaneState, useActiveItemPort">

    </Package>
  </Packages>
  <Theming href="/guides/theming">

  </Theming>
